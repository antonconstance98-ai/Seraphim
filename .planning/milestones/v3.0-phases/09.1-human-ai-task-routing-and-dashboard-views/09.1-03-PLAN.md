```typescript
import type { Decision } from '@/lib/types';

interface HumanTaskItem {
  id: string;         // decision id or synthetic id
  phase: string;
  taskId: string;     // from quality_signals.task_id if present, else phase
  name: string;       // from quality_signals.task_name if present, else phase
  description: string;
  status: 'pending' | 'in_progress' | 'done';
  timestamp: string;
}

interface AiTaskItem {
  id: number;
  phase: string;
  model: string;
  cost_usd: number;
  outcome: string;
  latency_ms: number;
  timestamp: string;
}

interface TaskListProps {
  type: 'human' | 'ai';
  humanTasks?: HumanTaskItem[];
  aiTasks?: AiTaskItem[];
}

export function TaskList({ type, humanTasks = [], aiTasks = [] }: TaskListProps) {
  if (type === 'human') {
    return (
      <div>
        {humanTasks.length === 0 ? (
          <p className="text-gray-500 text-sm">No pending human tasks.</p>
        ) : (
          <ul className="space-y-3">
            {humanTasks.map((task) => (
              <li key={task.id} className="bg-[#242424] border border-amber-500/20 rounded-lg p-4">
                <div className="flex items-start justify-between gap-3">
                  <div className="flex items-start gap-3">
                    <span className="text-amber-400 mt-0.5 text-sm">!</span>
                    <div>
                      <p className="text-gray-200 text-sm font-medium">
                        {task.name}
                      </p>
                      <p className="text-gray-500 text-xs mt-0.5 capitalize">{task.phase}</p>
                      {task.description && (
                        <p className="text-gray-400 text-xs mt-1">{task.description}</p>
                      )}
                    </div>
                  </div>
                  <span className="text-xs px-2 py-0.5 rounded bg-amber-500/15 text-amber-400 whitespace-nowrap">
                    {task.status}
                  </span>
                </div>
              </li>
            ))}
          </ul>
        )}
      </div>
    );
  }

  // AI tasks view
  return (
    <div>
      {aiTasks.length === 0 ? (
        <p className="text-gray-500 text-sm">No AI activity recorded.</p>
      ) : (
        <div className="overflow-hidden rounded-xl border border-white/10">
          <table className="w-full text-sm">
            <thead>
              <tr className="border-b border-white/10 text-gray-500 text-xs uppercase">
                <th className="text-left p-4">Phase</th>
                <th className="text-left p-4">Model</th>
                <th className="text-left p-4">Cost</th>
                <th className="text-left p-4">Outcome</th>
                <th className="text-left p-4">Latency</th>
              </tr>
            </thead>
            <tbody>
              {aiTasks.map((task) => (
                <tr key={task.id} className="border-b border-white/5 hover:bg-white/5">
                  <td className="p-4 text-gray-300 capitalize">{task.phase}</td>
                  <td className="p-4 text-gray-400 font-mono text-xs">{task.model}</td>
                  <td className="p-4 text-gray-300">${(task.cost_usd ?? 0).toFixed(4)}</td>
                  <td className="p-4">
                    <span className={`text-xs px-2 py-0.5 rounded ${
                      task.outcome === 'pass'
                        ? 'bg-green-500/15 text-green-400'
                        : task.outcome === 'human_required'
                        ? 'bg-amber-500/15 text-amber-400'
                        : task.outcome === 'loop_trigger'
                        ? 'bg-orange-500/15 text-orange-400'
                        : 'bg-gray-500/15 text-gray-400'
                    }`}>
                      {task.outcome}
                    </span>
                  </td>
                  <td className="p-4 text-gray-500 text-xs">{task.latency_ms}ms</td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      )}
    </div>
  );
}
```

# /seraphim:tasks

List human-assigned and AI-executed tasks for a pipeline phase, reading from blueprint.md and forge-log.md.

## Instructions

### Step 1: Resolve project root and phase-id

Find the project root by locating the nearest .seraphim/config.json (walk up from cwd).
If not found, abort: "No Seraphim project found. Run /seraphim:new-project first."

Extract phase-id from the command argument. If missing, abort:
"Usage: /seraphim:tasks [phase-id]
Example: /seraphim:tasks 01-auth-system"

### Step 2: Check prerequisites

Read .seraphim/phases/{phase-id}/blueprint.md. If missing:
"No blueprint found for phase {phase-id}. Run /seraphim:architect {phase-id} first."

Also check if .seraphim/phases/{phase-id}/forge-log.md exists (may not exist yet — that is OK).

### Step 3: Parse task markers from blueprint.md

```javascript
const { parseMarkers } = require('${CLAUDE_PLUGIN_ROOT}/lib/markers.js');
const blueprintContent = require('fs').readFileSync(blueprintPath, 'utf8');
const markers = parseMarkers(blueprintContent);
const taskMarkers = markers.filter(m => m.type === 'TASK');

const humanTasks = taskMarkers.filter(m => m.assignee === 'human');
const aiTasks = taskMarkers.filter(m => !m.assignee || m.assignee === 'ai');
```

### Step 4: Read execution status from forge-log.md (if exists)

If forge-log.md exists, parse SERAPHIM:FORGE_TASK markers to get execution status per task:

```javascript
const forgeContent = require('fs').readFileSync(forgeLogPath, 'utf8');
const forgeMarkers = parseMarkers(forgeContent);
const forgeTaskMap = {};
forgeMarkers.filter(m => m.type === 'FORGE_TASK').forEach(m => {
  forgeTaskMap[m.id] = m.status; // 'complete', 'failed', 'human_required', 'skipped'
});
```

If forge-log.md does not exist, forgeTaskMap is empty (all tasks show as 'not started').

### Step 5: Print task summary

Print the following format:

```
Phase: {phase-id}

HUMAN TASKS ({humanTasks.length})
──────────────────────────────────
{for each human task:}
  [{status icon}] {task.id}: {task.name || task.id}
       Type: {task.type || 'unspecified'}
       Status: {forgeTaskMap[task.id] || 'not started'}

AI TASKS ({aiTasks.length})
──────────────────────────────────
{for each AI task:}
  [{status icon}] {task.id}: {task.name || task.id}
       Type: {task.type || 'unspecified'}
       Status: {forgeTaskMap[task.id] || 'not started'}
```

Status icons:
  complete       → [x]
  failed         → [!]
  human_required → [H]
  skipped        → [-]
  not started    → [ ]

### Step 6: Print counts summary

```
Summary: {aiTasks.filter complete} AI tasks done, {humanTasks.length} human tasks pending
```

If there are human tasks, add:
```
Human tasks require your action before this phase is complete.
Run /seraphim:forge {phase-id} after completing them if remaining AI tasks exist.
```
