Follow the exact pattern from 02-RESEARCH.md Pattern 9:
```bash
#!/usr/bin/env bash
# websearch.sh — SearXNG web search wrapper for non-Claude models
# Usage: websearch.sh "query" [limit]
set -euo pipefail

QUERY="${1:?Usage: websearch.sh <query> [limit]}"
LIMIT="${2:-10}"
SEARXNG_URL="http://127.0.0.1:8888"   # per CLAUDE.md security: 127.0.0.1 not 0.0.0.0

curl -s --connect-timeout 5 --max-time 15 \
  "${SEARXNG_URL}/search?q=$(python3 -c "import urllib.parse,sys; print(urllib.parse.quote(sys.argv[1]))" "$QUERY")&format=json&pageno=1" \
  | python3 -c "
import json,sys
data = json.load(sys.stdin)
results = data.get('results', [])[:${LIMIT}]
for r in results:
    print(json.dumps({'title': r.get('title',''), 'url': r.get('url',''), 'content': r.get('content','')}))
"
```

**fetchdocs.js** — Create at `~/.claude/plugins/seraphim/tools/fetchdocs.js`:

Per research Open Question 1: Context7 may be MCP-only with no public REST API. Implement fetchdocs.js as a CLI tool that:
1. Accepts `<library-name> <query>` arguments
2. Attempts to call Context7 via `claude mcp call` subprocess (allows non-Claude models to access MCP indirectly through the Claude CLI)
3. If `claude` CLI is not available, falls back to a simple web search via websearch.sh with the library name + query
4. Outputs documentation text to stdout

```javascript
#!/usr/bin/env node
// fetchdocs.js — Context7 documentation fetcher for non-Claude models
// Usage: node fetchdocs.js <library-name> <query>
//
// Strategy:
//   1. Try: claude mcp call context7 resolve-library-id + query-docs (subprocess)
//   2. Fallback: websearch.sh "<library-name> <query> documentation"
//
// Note: Context7 is an MCP server with no confirmed public REST API.
// This script bridges MCP access for non-Claude models via claude CLI subprocess.
'use strict';

const { execSync } = require('child_process');
const path = require('path');

const libraryName = process.argv[2];
const query = process.argv[3];

if (!libraryName || !query) {
  console.error('Usage: node fetchdocs.js <library-name> <query>');
  process.exit(1);
}

// Strategy 1: Try Context7 via claude CLI
try {
  // Resolve library ID first
  const resolveResult = execSync(
    `echo '{"libraryName":"${libraryName.replace(/"/g, '\\"')}"}' | claude mcp call plugin__context7__context7 resolve-library-id 2>/dev/null`,
    { timeout: 15000, encoding: 'utf8' }
  );
  
  const libraries = JSON.parse(resolveResult);
  if (libraries && libraries.length > 0) {
    const libraryId = libraries[0].id || libraries[0].libraryId;
    if (libraryId) {
      const docsResult = execSync(
        `echo '{"context7CompatibleLibraryID":"${libraryId}","topic":"${query.replace(/"/g, '\\"')}"}' | claude mcp call plugin__context7__context7 query-docs 2>/dev/null`,
        { timeout: 30000, encoding: 'utf8' }
      );
      console.log(docsResult);
      process.exit(0);
    }
  }
} catch (e) {
  // claude CLI not available or MCP call failed — fall through to websearch
}

// Strategy 2: Fallback to websearch.sh
try {
  const websearchPath = path.join(__dirname, 'websearch.sh');
  const searchQuery = `${libraryName} ${query} documentation`;
  const results = execSync(`bash "${websearchPath}" "${searchQuery}" 5`, {
    timeout: 20000,
    encoding: 'utf8'
  });
  console.log(results);
} catch (e) {
  console.error('fetchdocs: both Context7 and websearch failed:', e.message);
  process.exit(1);
}
```
