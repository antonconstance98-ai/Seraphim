   **CRITICAL: Use `execFileSync` (not `execSync`) to avoid shell injection via file paths** (addresses review concern about shell metacharacters in paths). `execFileSync` takes the command and arguments as separate array elements, never passing through a shell:

   ```javascript
   const { execFileSync } = require('child_process');

   function getFileDiff(filePath, cwd) {
     try {
       // Check if HEAD exists (fresh repos have no commits)
       try {
         execFileSync('git', ['rev-parse', 'HEAD'], { cwd, timeout: 2000, stdio: 'pipe' });
       } catch (_) {
         // No HEAD -- try staged diff for initial commits
         const staged = execFileSync('git', ['diff', '--cached', '-U5', '--', filePath],
           { cwd, timeout: 5000, maxBuffer: 256 * 1024, stdio: 'pipe' }).toString();
         if (staged.trim()) return staged;
         // Fall through to untracked file handling below
       }

       // Case 1: tracked file with HEAD -- diff against HEAD
       const diff = execFileSync('git', ['diff', 'HEAD', '-U5', '--', filePath],
         { cwd, timeout: 5000, maxBuffer: 256 * 1024, stdio: 'pipe' }).toString();
       if (diff.trim()) return diff;

       // Case 2: untracked new file -- diff against empty tree
       try {
         execFileSync('git', ['diff', '--no-index', '/dev/null', filePath],
           { cwd, timeout: 5000, maxBuffer: 256 * 1024, stdio: 'pipe' });
       } catch (e) {
         // git diff --no-index exits non-zero when files differ; content is in stdout
         if (e.stdout) return e.stdout.toString();
       }
       return '';
     } catch (_) {
       return '';
     }
   }
   ```

   After obtaining the diff, check for `Binary files` marker: `if (diff.includes('Binary files')) return '';` -- skip binary content.

   - If count < threshold, check if ALL changed lines are non-meaningful:
     - Blank after stripping the `+`/`-` prefix
     - Comment lines: `/^(\/\/|#|\/\*|\*|--\s)/` (covers JS/TS/Python/Ruby/Go/Bash/SQL)
     - Pure whitespace changes
   - Do NOT classify string-literal-only lines as trivial (addresses review concern -- string changes like URLs, SQL, regexes can be security-relevant).
   - Return true (skip scan) only if ALL changed lines are blank, whitespace, or comments.
