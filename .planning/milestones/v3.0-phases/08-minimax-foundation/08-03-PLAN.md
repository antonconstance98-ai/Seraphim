    **Part B: Create `~/.claude/hooks/minimax-connectivity-test.js`**

    A standalone test script (NOT a hook) that verifies end-to-end MiniMax connectivity. Structure:

    ```javascript
    #!/usr/bin/env node
    // minimax-connectivity-test.js — Standalone connectivity verification for MiniMax M-2.7
    // Run: node ~/.claude/hooks/minimax-connectivity-test.js
    // Expected: Makes a real API call to api.minimax.io/v1, prints response details.

    'use strict';

    async function main() {
      // 1. Check MINIMAX_API_KEY is set
      if (!process.env.MINIMAX_API_KEY) {
        console.error('ERROR: MINIMAX_API_KEY environment variable is not set.');
        console.error('Set it with: export MINIMAX_API_KEY="your-key-here"');
        process.exit(1);
      }
      console.log('MINIMAX_API_KEY: set (' + process.env.MINIMAX_API_KEY.slice(0, 8) + '...)');

      // 2. Verify minimax-exec.js loads
      const { runMinimax } = require('./minimax-exec');
      console.log('minimax-exec.js: loaded');

      // 3. Verify codex-pricing.js has minimax entry
      const { CODEX_PRICING, computeCodexCostStrict } = require('./codex-pricing');
      const minimaxPricing = CODEX_PRICING['minimax-m2.7'];
      if (!minimaxPricing) {
        console.error('ERROR: minimax-m2.7 not found in CODEX_PRICING');
        process.exit(1);
      }
      console.log('codex-pricing.js: minimax-m2.7 entry present (input:' + minimaxPricing.input + ' output:' + minimaxPricing.output + ')');

      // 4. Make a real API call — simple prompt that should complete quickly
      console.log('\nCalling MiniMax M-2.7 API...');
      const startMs = Date.now();
      const result = await runMinimax('Reply with exactly: "MiniMax connectivity verified." Nothing else.', {
        maxTokens: 100,
        timeoutMs: 120000,  // 120s generous timeout for first call
      });
      const elapsed = Date.now() - startMs;

      // 5. Report results
      console.log('\n--- Results ---');
      console.log('success:', result.success);
      console.log('elapsed:', elapsed + 'ms');

      if (result.success) {
        console.log('text:', JSON.stringify(result.text));
        console.log('tokens:', JSON.stringify(result.tokens));
        console.log('cost: $' + result.cost);

        // Log the full response.usage to resolve Open Question #1 (cached token field name)
        console.log('\nVERIFICATION PASSED: MiniMax M-2.7 API is accessible and functional.');
      } else {
        console.error('error:', result.error);
        console.error('\nVERIFICATION FAILED: MiniMax API call failed.');
        process.exit(1);
      }
    }

    main().catch(e => { console.error('Unhandled:', e.message); process.exit(1); });
    ```
