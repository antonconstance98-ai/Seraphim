'use strict';
const fs = require('fs');
const path = require('path');

function readRoadmap(projectRoot) {
  const roadmapPath = path.join(projectRoot, '.seraphim', 'roadmap.json');
  try {
    if (fs.existsSync(roadmapPath)) {
      const raw = fs.readFileSync(roadmapPath, 'utf8');
      return JSON.parse(raw);
    }
  } catch (e) {}
  return { milestones: [] };
}

function writeRoadmap(projectRoot, data) {
  const seraphimDir = path.join(projectRoot, '.seraphim');
  if (!fs.existsSync(seraphimDir)) {
    fs.mkdirSync(seraphimDir, { recursive: true });
  }
  const roadmapPath = path.join(seraphimDir, 'roadmap.json');
  const tmpPath = path.join(seraphimDir, 'roadmap.json.tmp');
  fs.writeFileSync(tmpPath, JSON.stringify(data, null, 2), 'utf8');
  fs.renameSync(tmpPath, roadmapPath);
}

module.exports = { readRoadmap, writeRoadmap, findFeature };
