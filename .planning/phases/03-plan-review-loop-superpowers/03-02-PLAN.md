Add after the existing `computeCost` function (before `module.exports`):

```javascript
/**
 * runGpt54MiniApi(prompt, options)
 *
 * Direct OpenAI API call to GPT-5.4-mini for quick advisory responses.
 * Used by Superpowers parallel agent dispatch (SPWR-03) where CLI startup
 * overhead (1-2s) is too slow.
 *
 * @param {string} prompt       The prompt text
 * @param {object} [options]
 *   @param {string} [options.model]      Model ID (default: verified GPT-5.4-mini ID)
 *   @param {number} [options.maxTokens]  Max response tokens (default: 500)
 *   @param {number} [options.timeoutMs]  Abort timeout in ms (default: 15000)
 * @returns {Promise<object>}  { success, text, usage: {prompt_tokens, completion_tokens, total_tokens}, error }
 */
async function runGpt54MiniApi(prompt, options) {
  const opts = options || {};
  const modelId = opts.model || '{VERIFIED_MODEL_ID}'; // Use actual verified ID from Step 2
  const maxTokens = opts.maxTokens || 500;
  const timeoutMs = opts.timeoutMs || 15000;

  let OpenAI;
  try {
    OpenAI = require('openai').OpenAI;
  } catch (e) {
    return { success: false, text: '', usage: null, error: 'openai package not installed: ' + e.message };
  }

  const client = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);

  try {
    const response = await client.chat.completions.create({
      model: modelId,
      messages: [{ role: 'user', content: prompt }],
      max_tokens: maxTokens,
    }, { signal: controller.signal });

    return {
      success: true,
      text: response.choices[0].message.content,
      usage: response.usage, // { prompt_tokens, completion_tokens, total_tokens }
      error: null,
    };
  } catch (e) {
    return { success: false, text: '', usage: null, error: e.message };
  } finally {
    clearTimeout(timer);
  }
}
```

1. Read stdin JSON (SubagentStop payload)
2. Only process `hook_event_name === 'SubagentStop'`
3. Get cwd from `data.cwd || process.cwd()`
