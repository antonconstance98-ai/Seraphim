**app/api/events/route.ts** — Edge Runtime SSE:

CRITICAL: `export const runtime = 'edge'` — this is required. Vercel Serverless Functions close the connection after the function returns; Edge Runtime supports streaming without a hard timeout (per Pitfall 1 in RESEARCH.md). Do NOT use WebSocket. Do NOT use `export const runtime = 'nodejs'` on this file.

```typescript
export const runtime = 'edge';

export async function GET() {
  const encoder = new TextEncoder();

  const stream = new ReadableStream({
    async start(controller) {
      // Send initial heartbeat immediately
      controller.enqueue(encoder.encode(`data: ${JSON.stringify({ type: 'connected', ts: Date.now() })}\n\n`));

      // Poll for latest activity every 5 seconds
      // Query: SELECT MAX(last_pushed_at) AS latest FROM projects
      // Using getSql() from lib/db — import inside function body, not at module scope
      const { getSql } = await import('@/lib/db');
      const sql = getSql();

      let lastSeen: string | null = null;

      const interval = setInterval(async () => {
        try {
          const rows = await sql`SELECT MAX(last_pushed_at)::text AS latest FROM projects`;
          const latest = rows[0]?.latest ?? null;
          if (latest !== lastSeen) {
            lastSeen = latest;
            controller.enqueue(encoder.encode(
              `data: ${JSON.stringify({ type: 'update', latest_activity: latest, ts: Date.now() })}\n\n`
            ));
          }
        } catch {
          // Non-fatal — SSE continues even if a single poll fails
        }
      }, 5000);

      // Cleanup when client disconnects
      return () => clearInterval(interval);
    }
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache, no-transform',
      'Connection': 'keep-alive',
      'X-Accel-Buffering': 'no',
    },
  });
}
```

**hooks/useSSE.ts** ('use client' implied by usage in client components):

```typescript
'use client';
import { useEffect, useState } from 'react';

export interface SSEEvent {
  type: 'connected' | 'update';
  latest_activity?: string;
  ts: number;
}

export function useSSE(url: string) {
  const [connected, setConnected] = useState(false);
  const [lastEvent, setLastEvent] = useState<SSEEvent | null>(null);

  useEffect(() => {
    const es = new EventSource(url);

    es.onopen = () => setConnected(true);
    es.onerror = () => setConnected(false);
    es.onmessage = (e) => {
      try {
        const data = JSON.parse(e.data) as SSEEvent;
        setLastEvent(data);
        if (data.type === 'connected') setConnected(true);
      } catch { /* ignore malformed */ }
    };

    return () => es.close();
  }, [url]);

  return { connected, lastEvent };
}
```

**components/MetricsPanel.tsx** ('use client' — Chart.js requires DOM):

```typescript
'use client';
import { useEffect, useRef } from 'react';
import type { Decision } from '@/lib/types';

interface Props {
  decisions: Decision[];  // All decisions across all projects passed from server component
}

export function MetricsPanel({ decisions }: Props) {
  const costChartRef = useRef<HTMLCanvasElement>(null);
  const modelChartRef = useRef<HTMLCanvasElement>(null);

  useEffect(() => {
    if (!costChartRef.current || !modelChartRef.current) return;
    import('chart.js/auto').then(({ Chart }) => {
      // Cost trend: group by day, sum cost_usd
      // Compute from decisions prop — no DB call in client component
      const byDay = new Map<string, number>();
      for (const d of decisions) {
        const day = d.timestamp.slice(0, 10);
        byDay.set(day, (byDay.get(day) ?? 0) + Number(d.cost_usd));
      }
      const sortedDays = [...byDay.keys()].sort();

      const costChart = new Chart(costChartRef.current!, {
        type: 'line',
        data: {
          labels: sortedDays,
          datasets: [{
            label: 'Daily Cost (USD)',
            data: sortedDays.map(d => byDay.get(d) ?? 0),
            borderColor: '#6366f1',
            backgroundColor: 'rgba(99,102,241,0.15)',
            tension: 0.3,
            fill: true,
          }]
        },
        options: { responsive: true, plugins: { legend: { labels: { color: '#e5e7eb' } } }, scales: { x: { ticks: { color: '#9ca3af' } }, y: { ticks: { color: '#9ca3af' }, beginAtZero: true } } }
      });

      // Model performance: group by model, compute avg cost and success rate
      const byModel = new Map<string, { costs: number[]; successes: number; total: number }>();
      for (const d of decisions) {
        if (!byModel.has(d.model)) byModel.set(d.model, { costs: [], successes: 0, total: 0 });
        const m = byModel.get(d.model)!;
        m.costs.push(Number(d.cost_usd));
        m.total++;
        if (d.outcome === 'success') m.successes++;
      }
      const models = [...byModel.keys()];
      const avgCosts = models.map(m => {
        const e = byModel.get(m)!;
        return e.costs.reduce((a, b) => a + b, 0) / e.costs.length;
      });
      const successRates = models.map(m => {
        const e = byModel.get(m)!;
        return Math.round((e.successes / e.total) * 100);
      });

      new Chart(modelChartRef.current!, {
        type: 'bar',
        data: {
          labels: models.map(m => m.replace('claude-', '').replace('-4-6', '')),
          datasets: [
            { label: 'Avg Cost (USD)', data: avgCosts, backgroundColor: 'rgba(99,102,241,0.7)', yAxisID: 'y' },
            { label: 'Success Rate (%)', data: successRates, backgroundColor: 'rgba(52,211,153,0.7)', yAxisID: 'y1' }
          ]
        },
        options: {
          responsive: true,
          scales: {
            y: { type: 'linear', position: 'left', ticks: { color: '#9ca3af' } },
            y1: { type: 'linear', position: 'right', min: 0, max: 100, ticks: { color: '#9ca3af' } }
          },
          plugins: { legend: { labels: { color: '#e5e7eb' } } }
        }
      });

      return () => { costChart.destroy(); };
    });
  }, [decisions]);

  if (decisions.length === 0) {
    return <p className="text-gray-500 text-sm">No pipeline data yet. Run a Seraphim pipeline to populate metrics.</p>;
  }

  return (
    <div className="space-y-8">
      <div>
        <h3 className="text-sm font-medium text-gray-400 mb-3">Cost Trend (all projects)</h3>
        <canvas ref={costChartRef} height={120} />
      </div>
      <div>
        <h3 className="text-sm font-medium text-gray-400 mb-3">Model Performance</h3>
        <canvas ref={modelChartRef} height={120} />
      </div>
    </div>
  );
}
```

**hooks/phase-push.js** (CommonJS, 'use strict'):

This hook runs on the workstation inside the Claude Code hook system. It requires `./push-client` from the same `lib/` directory.

```javascript
'use strict';
const path = require('path');
const { pushProjectData } = require(path.join(__dirname, '..', 'lib', 'push-client'));

/**
 * phase-push.js — Claude Code hook
 * Fires after a Seraphim phase completes.
 * Reads stdin for Claude Code hook event JSON, extracts project root and phase ID,
 * then calls pushProjectData (fire-and-forget).
 */

let body = '';
process.stdin.setEncoding('utf8');
process.stdin.on('data', chunk => { body += chunk; });
process.stdin.on('end', () => {
  let event = {};
  try { event = JSON.parse(body); } catch { /* ignore parse errors */ }

  // Extract project root from hook event or fall back to cwd
  const projectRoot = event.cwd || process.cwd();

  // Extract phase ID from tool use context if available
  // Phase completion is signalled when a phase .md file is written (PostToolUse on Write tool)
  const toolInput = event.tool_input || {};
  const filePath = toolInput.file_path || '';

  // Detect which phase completed from the written file path
  const phaseMatch = filePath.match(/\.seraphim\/phases\/(\w+)\//);
  const phaseId = phaseMatch ? phaseMatch[1] : 'unknown';

  // Only push if this looks like a Seraphim phase output file
  // Avoid pushing on every single Write tool call
  const isPhaseOutput = /\.(seraphim\/phases\/)/.test(filePath) &&
    /\/(state\.json|forge-log\.md|crucible\.md|blueprint\.md|vision\.md|judgment\.md|external\.md|internal\.md)$/.test(filePath);

  if (!isPhaseOutput) {
    // Not a Seraphim phase output — exit silently
    process.stdout.write(JSON.stringify({ continue: true }));
    return;
  }

  // Fire-and-forget push — never blocks the hook return
  pushProjectData(projectRoot, phaseId);

  // Always continue — never block Claude
  process.stdout.write(JSON.stringify({ continue: true }));
});
```
