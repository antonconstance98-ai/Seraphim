```bash
PLUGIN_ROOT="$HOME/.claude/plugins/seraphim"
node -e "
  const fs = require('fs');
  const path = require('path');
  const decisionsPath = path.join('${PROJECT_ROOT}', '.seraphim', 'decisions.jsonl');

  if (!fs.existsSync(decisionsPath)) {
    console.log('NO_HISTORY');
    process.exit(0);
  }

  const content = fs.readFileSync(decisionsPath, 'utf8').trim();
  if (!content) {
    console.log('NO_HISTORY');
    process.exit(0);
  }

  const records = content.split('\n')
    .filter(l => l.trim())
    .map(l => {
      try { return JSON.parse(l); } catch(e) { return null; }
    })
    .filter(Boolean);

  // Group into runs: new 'discover' after non-discover starts a new run
  const runs = [];
  let current = [];
  for (const rec of records) {
    if (rec.phase === 'discover' && current.length > 0) {
      runs.push(current);
      current = [];
    }
    current.push(rec);
  }
  if (current.length > 0) runs.push(current);

  console.log(JSON.stringify(runs, null, 2));
"
```
