  <action>
    Create `~/.claude/hooks/migrate-opus-pricing.js` -- a standalone, idempotent migration script (per D-09).

    **Key insight from codebase analysis:** Per-project `token-log.jsonl` files do NOT contain `opus_baseline_usd` -- it is only in `global.jsonl`, added during aggregation by `computeOpusCost(rec.tokens)`. So the migration only needs to rewrite `~/.claude/dashboard/global.jsonl`.

    **Concurrency safety (addresses review concern):** The aggregator runs on SessionStart and APPENDS to global.jsonl via `fs.appendFileSync`. The migration rewrites the entire file. To prevent data loss from concurrent writes, the migration must:
    1. Create a lock file `~/.claude/dashboard/global.jsonl.lock` before reading. If the lock already exists, abort with a message to retry.
    2. Read the file, transform, write to temp, rename over original.
    3. Remove the lock file.
    This is a pragmatic guard for a single-user workstation. The aggregator does not check for locks (it appends), so the worst case with concurrent access is: aggregator appends a record after migration reads but before rename -- that record is lost from global.jsonl but will be re-added on next aggregation run (dedup key check will show it as missing). The lock file makes this window vanishingly small.

    **Full script implementation:**

    ```javascript
    #!/usr/bin/env node
    // migrate-opus-pricing.js -- One-time migration: recalculates opus_baseline_usd
    // in global.jsonl using corrected Opus 4.6 pricing ($5/$25 per Mtok).
    //
    // Idempotent: always recalculates from stored token counts. Running N times
    // produces byte-identical output (same JSON.stringify, same field order, same precision).
    //
    // Safe: uses lock file to prevent concurrent aggregator writes, atomic write
    // (temp-then-rename), and preserves ALL non-baseline fields byte-for-byte by
    // only modifying the opus_baseline_usd field on parsed records.
    //
    // After rewriting global.jsonl, regenerates dashboard.html via generateDashboard()
    // so derived artifacts reflect corrected savings percentages immediately.

    'use strict';
    const fs = require('fs');
    const path = require('path');
    const os = require('os');

    const DASHBOARD_DIR = path.join(os.homedir(), '.claude', 'dashboard');
    const GLOBAL_JSONL = path.join(DASHBOARD_DIR, 'global.jsonl');
    const LOCK_FILE = GLOBAL_JSONL + '.lock';

    // Corrected Opus 4.6 pricing (matches OPUS_PRICING in codex-pricing.js after D-07 fix)
    const OPUS_46 = { input: 5.00, cached_input: 1.25, output: 25.00 };

    function recalcOpusBaseline(tokens) {
      // Mirrors computeOpusCost exactly: no rounding, defaults to 0 for missing fields
      const inp    = (tokens && tokens.input)        || 0;
      const cached = (tokens && tokens.cached_input)  || 0;
      const out    = (tokens && tokens.output)        || 0;
      return (inp * OPUS_46.input + cached * OPUS_46.cached_input + out * OPUS_46.output) / 1_000_000;
    }

    function main() {
      // 1. Check source file exists
      if (!fs.existsSync(GLOBAL_JSONL)) {
        console.error('ERROR: ' + GLOBAL_JSONL + ' does not exist.');
        process.exit(1);
      }

      // 2. Acquire lock (concurrency guard against aggregator SessionStart hook)
      if (fs.existsSync(LOCK_FILE)) {
        console.error('ERROR: Lock file exists (' + LOCK_FILE + '). Another migration or aggregation may be running. Delete the lock file manually if stale, then retry.');
        process.exit(1);
      }
      fs.writeFileSync(LOCK_FILE, String(process.pid) + '\n', 'utf8');

      try {
        // 3. Read entire file
        const raw = fs.readFileSync(GLOBAL_JSONL, 'utf8');
        const lines = raw.split('\n');

        // 4. Process each line: preserve order, preserve blank lines, preserve malformed lines
        let processed = 0;
        let updated = 0;
        let malformed = 0;
        let blankLines = 0;
        let oldTotal = 0;
        let newTotal = 0;
        const outputLines = [];

        for (const line of lines) {
          // Preserve blank lines exactly (trailing newline handling)
          if (!line.trim()) {
            blankLines++;
            outputLines.push(line);
            continue;
          }

          let record;
          try {
            record = JSON.parse(line);
          } catch (e) {
            // Preserve malformed lines unchanged -- do not drop or modify
            malformed++;
            outputLines.push(line);
            continue;
          }

          processed++;

          // Only modify records that have tokens AND opus_baseline_usd
          if (record.tokens && record.opus_baseline_usd !== undefined) {
            oldTotal += record.opus_baseline_usd;
            record.opus_baseline_usd = recalcOpusBaseline(record.tokens);
            newTotal += record.opus_baseline_usd;
            updated++;
          } else if (record.tokens) {
            // Record has tokens but no opus_baseline_usd -- add it
            record.opus_baseline_usd = recalcOpusBaseline(record.tokens);
            newTotal += record.opus_baseline_usd;
            updated++;
          } else {
            // Record has no tokens field -- preserve unchanged
            if (record.opus_baseline_usd !== undefined) {
              oldTotal += record.opus_baseline_usd;
              newTotal += record.opus_baseline_usd;
            }
          }

          outputLines.push(JSON.stringify(record));
        }

        // 5. Atomic write: temp file then rename
        const tmpPath = GLOBAL_JSONL + '.tmp.' + process.pid;
        fs.writeFileSync(tmpPath, outputLines.join('\n'), 'utf8');
        fs.renameSync(tmpPath, GLOBAL_JSONL);

        // 6. Regenerate dashboard HTML to reflect corrected savings percentages
        //    (addresses review concern: derived artifacts must be rebuilt)
        let dashboardRegenerated = false;
        try {
          const { generateDashboard } = require('./codex-dashboard-generator');
          generateDashboard(DASHBOARD_DIR);
          dashboardRegenerated = true;
        } catch (e) {
          console.warn('WARNING: Dashboard regeneration failed: ' + e.message);
          console.warn('Run the aggregator manually to rebuild: node ~/.claude/hooks/codex-global-aggregator.js');
        }

        // 7. Print summary
        console.log('=== Opus Pricing Migration Summary ===');
        console.log('Records processed:   ' + processed);
        console.log('Records updated:     ' + updated);
        console.log('Malformed lines:     ' + malformed + ' (preserved unchanged)');
        console.log('Blank lines:         ' + blankLines + ' (preserved)');
        console.log('Old opus_baseline:   $' + oldTotal.toFixed(6));
        console.log('New opus_baseline:   $' + newTotal.toFixed(6));
        console.log('Reduction factor:    ' + (oldTotal > 0 ? (oldTotal / newTotal).toFixed(2) + 'x' : 'N/A'));
        console.log('Dashboard rebuilt:   ' + (dashboardRegenerated ? 'yes' : 'FAILED (see warning above)'));
        console.log('Idempotent:          yes (deterministic from token counts)');

      } finally {
        // 8. Release lock -- always, even on error
        try { fs.unlinkSync(LOCK_FILE); } catch (e) { /* ignore */ }
      }
    }

    main();
    ```

    **Idempotency definition (precise):** "Running the script N times on the same input produces byte-identical output each time, because `opus_baseline_usd` is always deterministically recomputed from the stored `tokens` field using fixed arithmetic, and all other fields are preserved via JSON parse-then-stringify of the same record object. No deduplication keys change. No fields are added or removed beyond `opus_baseline_usd`."

    **Record preservation guarantee:** The script processes each line independently. Blank lines are preserved as-is. Malformed (non-JSON) lines are preserved as-is. Valid JSON records have only `opus_baseline_usd` modified; all other fields are passed through `JSON.parse` then `JSON.stringify` which preserves values (though key ordering follows insertion order, which matches the original since we modify in-place on the parsed object).
