```typescript
// PM data upserts (optional fields -- backward compatible)
if (payload.milestones) {
  for (const m of payload.milestones) {
    await sql`
      INSERT INTO milestones (project_name, version, name, status, feature_count, complete_count, cost_usd)
      VALUES (${payload.project_name}, ${m.version}, ${m.name}, ${m.status}, ${m.feature_count}, ${m.complete_count}, ${m.cost_usd})
      ON CONFLICT (project_name, version) DO UPDATE
      SET name = EXCLUDED.name, status = EXCLUDED.status, feature_count = EXCLUDED.feature_count,
          complete_count = EXCLUDED.complete_count, cost_usd = EXCLUDED.cost_usd, synced_at = NOW()
    `;
  }
}

if (payload.features) {
  for (const f of payload.features) {
    await sql`
      INSERT INTO features (project_name, feature_id, slug, name, status, milestone_version, pipeline_phase, cost_usd)
      VALUES (${payload.project_name}, ${f.feature_id}, ${f.slug || null}, ${f.name}, ${f.status}, ${f.milestone_version}, ${f.pipeline_phase || null}, ${f.cost_usd})
      ON CONFLICT (project_name, feature_id) DO UPDATE
      SET slug = EXCLUDED.slug, name = EXCLUDED.name, status = EXCLUDED.status,
          milestone_version = EXCLUDED.milestone_version, pipeline_phase = EXCLUDED.pipeline_phase,
          cost_usd = EXCLUDED.cost_usd, synced_at = NOW()
    `;
  }
}

if (payload.human_tasks) {
  for (const t of payload.human_tasks) {
    await sql`
      INSERT INTO human_tasks (project_name, task_id, type, status, feature_id, urgency)
      VALUES (${payload.project_name}, ${t.task_id}, ${t.type}, ${t.status}, ${t.feature_id || null}, ${t.urgency})
      ON CONFLICT (project_name, task_id) DO UPDATE
      SET type = EXCLUDED.type, status = EXCLUDED.status, feature_id = EXCLUDED.feature_id,
          urgency = EXCLUDED.urgency, synced_at = NOW()
    `;
  }
}

if (payload.cost_trend) {
  // cost_trend entries have their own project_name per entry
  for (const entry of payload.cost_trend) {
    await sql`
      INSERT INTO cost_trend (project_name, date, cost_usd)
      VALUES (${entry.project_name}, ${entry.date}, ${entry.cost_usd})
      ON CONFLICT (project_name, date) DO UPDATE
      SET cost_usd = EXCLUDED.cost_usd
    `;
  }
}
```
