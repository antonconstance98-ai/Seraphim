```javascript
function pushPmData(projectRoot) {
  try {
    const r = require('./roadmap');
    const scanner = require('./multi-project-scanner');
    const roadmap = r.readRoadmap(projectRoot);
    const projectName = path.basename(projectRoot);

    // Build milestones array
    const milestones = (roadmap.milestones || []).map(m => {
      const prog = r.milestoneProgress(m);
      return {
        version: m.version, name: m.name, status: m.status,
        feature_count: prog.total, complete_count: prog.complete,
        cost_usd: 0  // cost aggregation from decisions.jsonl
      };
    });

    // Build features array
    const features = [];
    for (const m of (roadmap.milestones || [])) {
      for (const f of (m.features || [])) {
        features.push({
          feature_id: f.id, slug: f.slug || null, name: f.name,
          status: f.status, milestone_version: m.version,
          pipeline_phase: f.pipeline_phase || null, cost_usd: 0
        });
      }
    }

    // Build human_tasks array (scan task-completions.jsonl + forge-log.md)
    const human_tasks = scanPendingTasks(projectRoot, projectName);

    // Cost trend aggregation (D-06: across ALL projects by date)
    const allProjects = scanner.discoverSeraphimProjects([
      path.join(os.homedir(), 'projects'), path.join(os.homedir(), 'agent')
    ]);
    const cost_trend = aggregateCostByDate(allProjects.map(p => p.root));

    const payload = {
      project_name: projectName,
      root_path: projectRoot,
      profile: '',
      phase_id: '',
      phase_state: { phase_id: '', completed: false, loops: {}, retries: {} },
      decisions: [],
      pushed_at: new Date().toISOString(),
      milestones, features, human_tasks, cost_trend
    };

    const dashboardUrl = process.env.SERAPHIM_DASHBOARD_URL;
    if (!dashboardUrl) return;  // silently skip if no dashboard configured

    fetch(dashboardUrl + '/api/ingest', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer ' + (process.env.SERAPHIM_DASHBOARD_SECRET || '')
      },
      body: JSON.stringify(payload)
    }).catch(e => logError('PM push failed: ' + e.message));
  } catch (e) {
    logError('pushPmData error: ' + e.message);
  }
}
```

`aggregateCostByDate(projectRoots)` — reads decisions.jsonl from each project root (full file, not last 200 lines), groups by date (YYYY-MM-DD from timestamp), caps at last 365 days. Returns array of `{date, cost_usd, project_name}`.
