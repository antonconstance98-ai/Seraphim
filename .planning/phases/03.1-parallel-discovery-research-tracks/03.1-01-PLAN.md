**Step 6e — Merge all 4 track outputs into external.md.**

```bash
PLUGIN_ROOT="$HOME/.claude/plugins/seraphim"
node -e "
  const fs = require('fs');
  const m = require('${PLUGIN_ROOT}/lib/markers');
  const cfg = require('${PLUGIN_ROOT}/lib/config').read('${PROJECT_ROOT}');
  const now = new Date().toISOString();

  const tracks = [
    { n: 1, file: '/tmp/seraphim-track1-output.md', focus: 'solutions' },
    { n: 2, file: '/tmp/seraphim-track2-output.md', focus: 'anti-patterns' },
    { n: 3, file: '/tmp/seraphim-track3-output.md', focus: 'architecture' },
    { n: 4, file: '/tmp/seraphim-track4-output.md', focus: 'dependencies' }
  ];

  const trackContents = tracks.map(t => {
    const content = fs.existsSync(t.file) ? fs.readFileSync(t.file, 'utf8').trim() : '';
    return content || m.emitTrackFailed('unknown', 'output file missing') + '\n\n## Track ' + t.n + ': TRACK_FAILED\n\nNo output produced.';
  });

  const failedCount = trackContents.filter(c => c.includes('TRACK_FAILED')).length;
  const status = failedCount === 4 ? 'all_failed' : failedCount > 0 ? 'partial' : 'complete';

  const header = m.emitPhaseStart('discover_external', 'RESOLVED_MODEL_ID', cfg.profile);
  const footer = m.emitPhaseComplete({ phase: 'discover_external', model: 'RESOLVED_MODEL_ID', status, tracks_completed: String(4 - failedCount), tracks_failed: String(failedCount) });

  const body = [
    '# External Research: ${PHASE_DESCRIPTION}',
    '',
    '_Parallel research across 4 focused tracks. Each track used the profile\\'s discover_external model._',
    '',
    ...trackContents,
    ''
  ].join('\n');

  const output = header + '\n\n' + body + '\n' + footer + '\n';
  fs.writeFileSync('${PROJECT_ROOT}/.seraphim/phases/${PHASE_ID}/discovery/external.md', output, 'utf8');
  console.log('external.md written. Status:', status, '| Tracks completed:', 4 - failedCount, '| Tracks failed:', failedCount);
"
```
