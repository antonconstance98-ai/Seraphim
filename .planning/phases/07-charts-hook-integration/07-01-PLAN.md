Change that closing sequence to:

```javascript
// Initialize on load
renderSessionTable(1);

// ── Chart.js initialization (CHART-01, CHART-02, CHART-03) ──────────────────
(function() {
  if (typeof Chart === 'undefined') return; // Chart.js not available — skip silently

  var grouping = 'daily';
  var costSavingsChartInst = null;

  // ── Helpers ────────────────────────────────────────────────────────────────

  function groupTimeSeries(mode) {
    var raw = DASHBOARD_DATA.timeSeries || [];
    if (mode === 'daily') return raw;

    // Weekly grouping: ISO week key YYYY-WNN
    var buckets = {};
    for (var i = 0; i < raw.length; i++) {
      var d = raw[i];
      var dt = new Date(d.date + 'T00:00:00Z');
      // ISO week: Thursday-based week number
      var jan4 = new Date(Date.UTC(dt.getUTCFullYear(), 0, 4));
      var startOfWeek1 = new Date(jan4);
      startOfWeek1.setUTCDate(jan4.getUTCDate() - ((jan4.getUTCDay() + 6) % 7));
      var weekNum = Math.floor((dt - startOfWeek1) / 604800000) + 1;
      var key = dt.getUTCFullYear() + '-W' + String(weekNum).padStart(2, '0');
      if (!buckets[key]) buckets[key] = { date: key, cost: 0, opusBaseline: 0 };
      buckets[key].cost += d.cost;
      buckets[key].opusBaseline += d.opusBaseline;
    }
    return Object.values(buckets).sort(function(a, b) { return a.date < b.date ? -1 : a.date > b.date ? 1 : 0; });
  }

  function buildCostSavingsChart(mode) {
    var canvas = document.getElementById('costSavingsChart');
    if (!canvas) return;
    var ctx = canvas.getContext('2d');
    if (!ctx) return;

    var series = groupTimeSeries(mode);
    var labels = series.map(function(d) { return d.date; });
    var costs   = series.map(function(d) { return d.cost; });
    var baselines = series.map(function(d) { return d.opusBaseline; });

    if (costSavingsChartInst) { costSavingsChartInst.destroy(); }

    costSavingsChartInst = new Chart(ctx, {
      type: 'line',
      data: {
        labels: labels,
        datasets: [
          {
            label: 'Actual Cost ($)',
            data: costs,
            borderColor: '#00d4ff',
            backgroundColor: 'rgba(0,212,255,0.08)',
            fill: false,
            tension: 0.3,
            pointRadius: 3
          },
          {
            label: 'Opus Baseline ($)',
            data: baselines,
            borderColor: '#ffab00',
            backgroundColor: 'rgba(255,171,0,0.08)',
            fill: false,
            tension: 0.3,
            pointRadius: 3,
            borderDash: [5, 3]
          }
        ]
      },
      options: {
        responsive: true,
        plugins: {
          legend: { labels: { color: '#e0e0e0' } }
        },
        scales: {
          x: {
            ticks: { color: '#a0a0b0', maxTicksLimit: 12 },
            grid:  { color: '#2a2a4a' }
          },
          y: {
            ticks: {
              color: '#a0a0b0',
              callback: function(v) { return '$' + v.toFixed(4); }
            },
            grid: { color: '#2a2a4a' }
          }
        }
      }
    });
  }

  function buildProjectBarChart() {
    var canvas = document.getElementById('projectBarChart');
    if (!canvas) return;
    var ctx = canvas.getContext('2d');
    if (!ctx) return;

    // Sort by savingsPct descending, exclude Unattributed
    var rows = (DASHBOARD_DATA.projectTable || [])
      .filter(function(r) { return r.projectName !== 'Unattributed'; })
      .slice()
      .sort(function(a, b) { return parseFloat(b.savingsPct) - parseFloat(a.savingsPct); });

    var labels = rows.map(function(r) { return r.projectName; });
    var savings = rows.map(function(r) { return parseFloat(r.savingsPct); });

    new Chart(ctx, {
      type: 'bar',
      data: {
        labels: labels,
        datasets: [{
          label: 'Savings %',
          data: savings,
          backgroundColor: 'rgba(0,230,118,0.7)',
          borderColor: '#00e676',
          borderWidth: 1
        }]
      },
      options: {
        indexAxis: 'y',
        responsive: true,
        plugins: {
          legend: { labels: { color: '#e0e0e0' } }
        },
        scales: {
          x: {
            ticks: {
              color: '#a0a0b0',
              callback: function(v) { return v + '%'; }
            },
            grid: { color: '#2a2a4a' }
          },
          y: {
            ticks: { color: '#a0a0b0' },
            grid: { color: '#2a2a4a' }
          }
        }
      }
    });
  }

  function setChartGrouping(mode) {
    grouping = mode;
    document.getElementById('btn-daily').style.color  = mode === 'daily'  ? 'var(--text-accent)' : 'var(--text-secondary)';
    document.getElementById('btn-weekly').style.color = mode === 'weekly' ? 'var(--text-accent)' : 'var(--text-secondary)';
    document.getElementById('btn-daily').style.background  = mode === 'daily'  ? 'var(--bg-card)'      : 'var(--bg-secondary)';
    document.getElementById('btn-weekly').style.background = mode === 'weekly' ? 'var(--bg-card)'      : 'var(--bg-secondary)';
    buildCostSavingsChart(mode);
  }
  // Expose toggle to inline onclick handlers
  window.setChartGrouping = setChartGrouping;

  // Initialize charts on load
  buildCostSavingsChart(grouping);
  buildProjectBarChart();
})();
</script>
```
