```
// ── Chart.js initialization (CHART-01, CHART-02, CHART-03) ──────────────────
// Entire block is guarded: if Chart.js is absent (chart.min.js unavailable),
// the IIFE returns immediately — no exceptions, tables/sections remain intact.
(function() {
  if (typeof Chart === 'undefined') return;

  var grouping = 'daily';
  var costSavingsChartInst = null;

  // groupTimeSeries(mode) — returns daily or ISO-week-bucketed array.
  // Weekly algorithm: ISO 8601 Thursday-based weeks (YYYY-WNN keys), UTC dates.
  // Handles empty input: returns [] without error.
  function groupTimeSeries(mode) {
    var raw = DASHBOARD_DATA.timeSeries || [];
    if (mode === 'daily') return raw;
    var buckets = {};
    for (var i = 0; i < raw.length; i++) {
      var entry = raw[i];
      var dt = new Date(entry.date + 'T00:00:00Z');
      if (isNaN(dt.getTime())) continue; // skip malformed date strings
      var jan4 = new Date(Date.UTC(dt.getUTCFullYear(), 0, 4));
      var mondayOfWeek1 = new Date(jan4);
      mondayOfWeek1.setUTCDate(jan4.getUTCDate() - ((jan4.getUTCDay() + 6) % 7));
      var weekNum = Math.floor((dt - mondayOfWeek1) / 604800000) + 1;
      var key = dt.getUTCFullYear() + '-W' + String(weekNum).padStart(2, '0');
      if (!buckets[key]) buckets[key] = { date: key, cost: 0, opusBaseline: 0 };
      buckets[key].cost += (typeof entry.cost === 'number' && isFinite(entry.cost) ? entry.cost : 0);
      buckets[key].opusBaseline += (typeof entry.opusBaseline === 'number' && isFinite(entry.opusBaseline) ? entry.opusBaseline : 0);
    }
    return Object.values(buckets).sort(function(a, b) { return a.date < b.date ? -1 : a.date > b.date ? 1 : 0; });
  }

  function buildCostSavingsChart(mode) {
    var canvas = document.getElementById('costSavingsChart');
    if (!canvas) return;
    var ctx = canvas.getContext('2d');
    if (!ctx) return;
    var series = groupTimeSeries(mode);
    var labels    = series.map(function(e) { return e.date; });
    var costs     = series.map(function(e) { return e.cost; });
    var baselines = series.map(function(e) { return e.opusBaseline; });
    if (costSavingsChartInst) { costSavingsChartInst.destroy(); }
    costSavingsChartInst = new Chart(ctx, {
      type: 'line',
      data: {
        labels: labels,
        datasets: [
          {
            label: 'Actual Cost ($)',
            data: costs,
            borderColor: '#00d4ff',
            backgroundColor: 'rgba(0,212,255,0.08)',
            fill: false,
            tension: 0.3,
            pointRadius: series.length < 30 ? 3 : 0
          },
          {
            label: 'Opus Baseline ($)',
            data: baselines,
            borderColor: '#ffab00',
            backgroundColor: 'rgba(255,171,0,0.08)',
            fill: false,
            tension: 0.3,
            pointRadius: series.length < 30 ? 3 : 0,
            borderDash: [5, 3]
          }
        ]
      },
      options: {
        responsive: true,
        plugins: { legend: { labels: { color: '#e0e0e0' } } },
        scales: {
          x: { ticks: { color: '#a0a0b0', maxTicksLimit: 12 }, grid: { color: '#2a2a4a' } },
          y: {
            ticks: { color: '#a0a0b0', callback: function(v) { return '$' + v.toFixed(4); } },
            grid: { color: '#2a2a4a' }
          }
        }
      }
    });
  }

  function buildProjectBarChart() {
    var canvas = document.getElementById('projectBarChart');
    if (!canvas) return;
    var ctx = canvas.getContext('2d');
    if (!ctx) return;
    // savingsPct is a string (e.g. "72.3") from computeMetrics — parseFloat handles it.
    // Exclude Unattributed; treat NaN savingsPct as 0 for sort stability.
    var rows = (DASHBOARD_DATA.projectTable || [])
      .filter(function(r) { return r.projectName !== 'Unattributed'; })
      .slice()
      .sort(function(a, b) {
        var av = parseFloat(a.savingsPct); if (isNaN(av)) av = 0;
        var bv = parseFloat(b.savingsPct); if (isNaN(bv)) bv = 0;
        return bv - av;
      });
    if (rows.length === 0) return; // nothing to chart — hide canvas gracefully
    var labels  = rows.map(function(r) { return r.projectName; });
    var savings = rows.map(function(r) { var v = parseFloat(r.savingsPct); return isNaN(v) ? 0 : v; });
    new Chart(ctx, {
      type: 'bar',
      data: {
        labels: labels,
        datasets: [{
          label: 'Savings %',
          data: savings,
          backgroundColor: 'rgba(0,230,118,0.7)',
          borderColor: '#00e676',
          borderWidth: 1
        }]
      },
      options: {
        indexAxis: 'y',
        responsive: true,
        plugins: { legend: { labels: { color: '#e0e0e0' } } },
        scales: {
          x: { ticks: { color: '#a0a0b0', callback: function(v) { return v + '%'; } }, grid: { color: '#2a2a4a' } },
          y: { ticks: { color: '#a0a0b0' }, grid: { color: '#2a2a4a' } }
        }
      }
    });
  }

  function setChartGrouping(mode) {
    grouping = mode;
    var btnDaily  = document.getElementById('btn-daily');
    var btnWeekly = document.getElementById('btn-weekly');
    if (btnDaily)  { btnDaily.style.color  = mode === 'daily'  ? 'var(--text-accent)'   : 'var(--text-secondary)'; btnDaily.style.background  = mode === 'daily'  ? 'var(--bg-card)'    : 'var(--bg-secondary)'; }
    if (btnWeekly) { btnWeekly.style.color = mode === 'weekly' ? 'var(--text-accent)'   : 'var(--text-secondary)'; btnWeekly.style.background = mode === 'weekly' ? 'var(--bg-card)'    : 'var(--bg-secondary)'; }
    buildCostSavingsChart(mode);
  }
  window.setChartGrouping = setChartGrouping;

  buildCostSavingsChart(grouping);
  buildProjectBarChart();
})();
</script>
```
