const { computeCostForModel } = require('${CLAUDE_PLUGIN_ROOT}/lib/pricing.js');

// Sum costs from both passes (verify + adversarial)
// verifyUsage and adversarialUsage should be captured from executor return values
const verifyCost = verifyUsage
  ? computeCostForModel(verifyModel, { input_tokens: verifyUsage.input_tokens || 0, output_tokens: verifyUsage.output_tokens || 0 })

  ? computeCostForModel(adversarialModel, { input_tokens: adversarialUsage.input_tokens || 0, output_tokens: adversarialUsage.output_tokens || 0 })

const totalCost = verifyCost + adversarialCost;
const totalTokensIn = (verifyUsage ? verifyUsage.input_tokens || 0 : 0) + (adversarialUsage ? adversarialUsage.input_tokens || 0 : 0);
const totalTokensOut = (verifyUsage ? verifyUsage.output_tokens || 0 : 0) + (adversarialUsage ? adversarialUsage.output_tokens || 0 : 0);

const record = buildRecord({

  tokens_out: totalTokensOut,
