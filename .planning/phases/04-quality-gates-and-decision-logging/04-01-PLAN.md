```javascript
'use strict';
const fs = require('fs');
const path = require('path');

const REQUIRED_FIELDS = ['timestamp', 'phase', 'model', 'profile', 'tokens_in', 'tokens_out', 'cost_usd', 'latency_ms', 'outcome'];

function appendDecision(projectRoot, record) {
  const decisionsPath = path.join(projectRoot, '.seraphim', 'decisions.jsonl');
  const dir = path.dirname(decisionsPath);
  try {
    if (!fs.existsSync(dir)) fs.mkdirSync(dir, { recursive: true });
    const line = JSON.stringify(record) + '\n';
    fs.appendFileSync(decisionsPath, line, 'utf8');
  } catch (e) {
    process.stderr.write(`[decisions-logger] Failed to append: ${e.message}\n`);
  }
}

function buildRecord({ phase, model, profile, tokens_in, tokens_out, cost_usd, latency_ms, outcome, retry_count = 0, loop_count = 0, quality_signals = {} }) {
  return {
    timestamp: new Date().toISOString(),
    phase,
    model,
    profile,
    tokens_in: tokens_in || 0,
    tokens_out: tokens_out || 0,
    cost_usd: typeof cost_usd === 'number' ? cost_usd : 0,
    latency_ms: latency_ms || 0,
    outcome,  // 'success' | 'failure' | 'partial'
    retry_count,
    loop_count,
    quality_signals: Object.assign({
      crucible_pass_rate: null,
      judge_kill_rate: null,
      checkpoint_catch_rate: null,
      loop_trigger_reason: null
    }, quality_signals)
  };
}

module.exports = { appendDecision, buildRecord, REQUIRED_FIELDS };
```

```javascript
'use strict';
const fs = require('fs');
const path = require('path');
const { REQUIRED_FIELDS } = require('./decisions-logger.js');

function validateDecisions(projectRoot) {
  const decisionsPath = path.join(projectRoot, '.seraphim', 'decisions.jsonl');
  if (!fs.existsSync(decisionsPath)) return { valid: true, violations: [] };

  let lines;
  try {
    lines = fs.readFileSync(decisionsPath, 'utf8').split('\n').filter(l => l.trim());
  } catch (e) {
    return { valid: false, violations: [{ line: 0, error: `Cannot read file: ${e.message}` }] };
  }

  const violations = [];
  lines.forEach((line, idx) => {
    let record;
    try { record = JSON.parse(line); }
    catch (e) {
      violations.push({ line: idx + 1, error: 'JSON parse failure', raw: line.slice(0, 80) });
      return;
    }
    REQUIRED_FIELDS.forEach(field => {
      if (record[field] === undefined || record[field] === null) {
        violations.push({ line: idx + 1, field, error: 'missing required field' });
      }
    });
    if (typeof record.cost_usd === 'number' && record.cost_usd < 0) {
      violations.push({ line: idx + 1, field: 'cost_usd', error: 'negative cost', value: record.cost_usd });
    }
    if (typeof record.tokens_in === 'number' && record.tokens_in < 0) {
      violations.push({ line: idx + 1, field: 'tokens_in', error: 'negative token count', value: record.tokens_in });
    }
    if (typeof record.tokens_out === 'number' && record.tokens_out < 0) {
      violations.push({ line: idx + 1, field: 'tokens_out', error: 'negative token count', value: record.tokens_out });
    }
    if ((record.tokens_in > 10_000_000) || (record.tokens_out > 10_000_000)) {
      violations.push({ line: idx + 1, error: 'impossible token count', tokens_in: record.tokens_in, tokens_out: record.tokens_out });
    }
  });

  return { valid: violations.length === 0, violations };
}

function reportViolations(violations) {
  if (violations.length === 0) return 'decisions.jsonl: no violations found.';
  const lines = [`decisions.jsonl integrity check: ${violations.length} violation(s) found:`];
  violations.forEach(v => {
    const loc = `line ${v.line}${v.field ? `, field '${v.field}'` : ''}`;
    const val = v.value !== undefined ? ` (value: ${v.value})` : '';
    lines.push(`  [${loc}] ${v.error}${val}`);
  });
  return lines.join('\n');
}

module.exports = { validateDecisions, reportViolations };
```
