### Step 2b: Detect crucible fix mode (QUAL-04)

Check if crucible.md exists for this phase AND has a fail verdict:

```javascript
const cruciblePath = path.join(projectRoot, '.seraphim', 'phases', phaseId, 'crucible.md');
let fixMode = false;
let fixTaskIds = new Set();
let fixInstructions = {};

if (fs.existsSync(cruciblePath)) {
  const crucibleContent = fs.readFileSync(cruciblePath, 'utf8');
  const { parseMarkers } = require('${CLAUDE_PLUGIN_ROOT}/lib/markers.js');
  const crucibleMarkers = parseMarkers(crucibleContent);
  const phaseComplete = crucibleMarkers.find(m => m.type === 'PHASE_COMPLETE' && m.phase === 'crucible');
  
  if (phaseComplete && phaseComplete.verdict === 'fail') {
    fixMode = true;
    console.log('Crucible fix mode activated — Crucible returned verdict=fail.');
    console.log('Forge will re-run only the tasks referenced in Crucible issues.');
    
    // Extract task IDs from the ## Issues Found section
    // Look for task ID references in the crucible.md prose (format: "task-{id}" or "Task {id}")
    const taskIdRe = /\btask[-\s](\S+)/gi;
    let m;
    while ((m = taskIdRe.exec(crucibleContent)) !== null) {
      fixTaskIds.add(m[1].replace(/[.,;:)]/g, ''));
    }
    
    // Extract fix instructions block if present
    const fixBlockMatch = crucibleContent.match(/##\s+Fix Instructions\s*\n([\s\S]*?)(?=\n##|\n<!--|$)/i);
    if (fixBlockMatch) {
      fixInstructions['_global'] = fixBlockMatch[1].trim();
    }
    
    if (fixTaskIds.size === 0) {
      console.log('Warning: Crucible is in fail mode but no specific task IDs found in issues. Running all tasks.');
      fixMode = false;
    }
  }
}
```
